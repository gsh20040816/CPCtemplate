\section{素因子估值与升幂：先约去公共因子}\label{knowledge-lte}
\index{p-adic valuation}\index{lifting the exponent}\index{LTE}\index{升幂引理}
$p$ 为素数，$a,b,n$ 为正整数，$v_p(x)$ 为非零整数 $x$ 中 $p$ 的重数。
\textbf{$v_p(0)$ 无定义}：差式先判 $a=b$。

\subsection{基本运算与公共因子}
$v_p(xy)=v_p(x)+v_p(y)$，$v_p(x^n)=nv_p(x)$；$v_p(x+y)\ge\min(v_p(x),v_p(y))$，估值不同才取等。
记 $\alpha=v_p(a)$、$\beta=v_p(b)$：若 $\alpha\ne\beta$，$v_p(a^n\pm b^n)=n\min(\alpha,\beta)$；
若 $\alpha=\beta=t$，写 $a=p^tA,b=p^tB$，则 $v_p(a^n\pm b^n)=nt+v_p(A^n\pm B^n)$，再对 $A,B$ 用 LTE。

\subsection{奇素数：差式与和式的条件不同}
$p$ 为奇素数且 $p\nmid ab$：
\[
\begin{aligned}
p\mid a-b,\ a\ne b
&\quad\Longrightarrow\quad v_p(a^n-b^n)=v_p(a-b)+v_p(n),\\
p\mid a+b,\ n\text{ 为奇数}
&\quad\Longrightarrow\quad v_p(a^n+b^n)=v_p(a+b)+v_p(n).
\end{aligned}
\]
差式不要求 $n$ 奇，和式必须 $n$ 奇；不要求 $\gcd(a,b)=1$。

\subsection{素数 2：先分指数奇偶}
$a,b$ 均为奇数，差式要求 $a\ne b$：
\[
v_2(a^n-b^n)=
\begin{cases}
v_2(a-b),&n\text{ 奇},\\
v_2(a-b)+v_2(a+b)+v_2(n)-1,&n\text{ 偶},
\end{cases}\qquad
v_2(a^n+b^n)=
\begin{cases}
v_2(a+b),&n\text{ 奇},\\
1,&n\text{ 偶}.
\end{cases}
\]
若 $4\mid a-b$，差式可写 $v_2(a-b)+v_2(n)$。奇偶不同则和差均奇；都偶先约去公共 2 因子。

\subsection{把大整数整除转成指数条件}
$M=\prod p_i^{e_i}\mid X$（$X\ne0$）等价于逐项 $v_{p_i}(X)\ge e_i$，无需构造大幂。
LTE 只处理满足条件的局部因子，不能代替乘法阶或离散对数。$a+b$、$|a-b|$、$nt$ 须防溢出。
阶乘估值用 $v_p(N!)=\sum_{j\ge1}\lfloor N/p^j\rfloor$（ExLucas，第~\pageref{compact-ExLucas}~页），不套 LTE；
素数幂开根还须检查单位部分有根，估值条件不充分。

\section{整除与反演：先辨认求和方向}\label{knowledge-mobius-model}
\index{Dirichlet convolution}\index{Mobius inversion}\index{GCD 加权求和}
数论函数定义在\textbf{正整数}上；下标 0 只是前缀和占位。
\[
(f*g)(n)=\sum_{d\mid n}f(d)g(n/d),\qquad
\varepsilon(n)=[n=1],\quad \mathbf1(n)=1,\quad \operatorname{id}(n)=n,
\qquad \mu*\mathbf1=\varepsilon,\quad \varphi*\mathbf1=\operatorname{id}.
\]

\subsection{约数和与倍数和不能混用}
\textbf{约数方向。}$F(n)=\sum_{d\mid n}f(d)\Rightarrow f(n)=\sum_{d\mid n}\mu(d)F(n/d)$，不要求 $f$ 积性。

\textbf{倍数方向。}$E(k)$ 仅在 $1\le k\le V$ 非零，$G(d)=\sum_{j\ge1,\,dj\le V}E(dj)$，则
\[
E(k)=\sum_{t=1}^{\lfloor V/k\rfloor}\mu(t)G(kt),
\]
或从大到小 $E(k)=G(k)-\sum_{j\ge2,\,kj\le V}E(kj)$。须有限支持。
互质下标对：$c_d=\#\{i:d\mid a_i\}$，答案 $\sum_{d=1}^V\mu(d)\binom{c_d}{2}$（重复值按下标区分）。
枚举倍数求 $c_d$，时间 $O(V\log V)$、空间 $O(V)$；$V$ 太大时须改为枚举实际约数。

\subsection{GCD 权值先反演，再交换求和}
令 $h=\mu*w$（$w=h*\mathbf1$），则
\[
\sum_{x=1}^A\sum_{y=1}^B w(\gcd(x,y))
=\sum_{d=1}^{\min(A,B)}h(d)
\left\lfloor\frac Ad\right\rfloor\left\lfloor\frac Bd\right\rfloor.
\]
互质计数 $h=\mu$，GCD 总和 $h=\varphi$；系数是卷积 $h(d)$，一般不是 $\mu(d)w(d)$。
已知 $h$ 的前缀和后按第~\ref{knowledge-quotient-choice}~节分块。

\section{GCD 计数：缩放、边界与对称性}\label{knowledge-gcd-domains}
\index{互质数对}\index{无序数对}
记 $C(A,B)$ 为 $[1,A]\times[1,B]$ 中的\textbf{有序}互质数对数，任一边为 0 时为 0；
不含坐标轴，不去掉 $x=y$。

\subsection{恰好等于 \texorpdfstring{$k$}{k}：先缩放整除对象，再做矩形容斥}
$k\ge1$，$a,c\ge1$，$a\le x\le b$、$c\le y\le d$ 中 $\gcd=k$ 的对数为
$C(A,B)-C(A_0,B)-C(A,B_0)+C(A_0,B_0)$，其中
\[
A=\lfloor b/k\rfloor,\quad A_0=\lfloor(a-1)/k\rfloor,\qquad
B=\lfloor d/k\rfloor,\quad B_0=\lfloor(c-1)/k\rfloor.
\]
先减 1 再除；$k>\min(b,d)$ 时自动为 0。多测预筛到所有查询的 $\max\min(b/k,d/k)$。

\subsection{去掉顺序之前，先检查交换是否仍在定义域内}
同一正整数区间 $I$，$T$ 为 $I\times I$ 中 GCD 为 $k$ 的有序对数，$\delta=[k\in I]$：
$\#\{x<y\}=(T-\delta)/2$，$\#\{x\le y\}=(T+\delta)/2$。
仅当定义域交换对称时可用；一般矩形不能直接除以 2。不同下标计数回到频率反演。
正方形：$C(n,n)=2\Phi(n)-1$，$\Phi(n)=\sum_{i\le n}\varphi(i)$，$n\ge1$；$n=0$ 为 0。
含 0：$\gcd(0,y)=|y|$，轴上仅 $(0,k),(k,0)$ 另行判断；$(0,0)$ 按题面约定；负数处理绝对值。

\section{整除分块：前缀和来源决定算法}\label{knowledge-quotient-choice}
\index{floor quotient grouping}\index{筛法选型}
$H(t)=\sum_{i\le t}h(i)$，$H(0)=0$，$q_A=\lfloor A/l\rfloor,q_B=\lfloor B/l\rfloor$，$l\le\min(A,B)$：
\[
r=\min\!\left(\left\lfloor A/q_A\right\rfloor,
                 \left\lfloor B/q_B\right\rfloor\right),
\quad\text{整块贡献 }(H(r)-H(l-1))q_Aq_B.
\]
两边都要参与取 $r$。段数 $O(\sqrt A+\sqrt B)$（且 $\le\min(A,B)$），不是 $O(\sqrt{\min(A,B)})$。

\subsection{按数据与资源选前缀和}
\begin{description}
\item[上界可预筛，询问多：]
第~\pageref{compact-LinearSieve}~页 \texttt{LinearSieve s(L)} 的 \texttt{s.mu}、\texttt{s.phi} 建前缀和，
$O(L)$ 时间空间，查表 $O(1)$；筛上限覆盖短边。$\mu$ 前缀可负，$\varphi$ 前缀勿用 \texttt{int}。
\item[只求少量巨大前缀：]
第~\pageref{compact-DuJiao}~页 \texttt{DuJiao s(B)}：\texttt{s.mertens(n)}（有符号 64 位）、
\texttt{s.totient\_sum(n)}（128 位），$0\le n\le10^{11}$，$1\le B<\texttt{INT\_MAX}$，$B$ 是预筛上限。
单次时间 $O(B+N/\sqrt B)$，$B\approx N^{2/3}$ 得 $O(N^{2/3})$（哈希期望）。
\end{description}
内存：\texttt{DuJiao} 两张 64 位前缀表约 $16(B+1)$ 字节，构造时另有 \texttt{LinearSieve} 表；
$N=10^{11}$ 取 $B=N^{2/3}$ 约 345 MB，先看内存限制。不同查询增缓存；勿每次重建筛对象。
正方形互质对直接 \texttt{2 * s.totient\_sum(n) - 1}（$n\ge1$）。
一般矩形用 \texttt{mertens} 分块时，内层递归使总复杂度不止外层段数。
\[
M(n)=1-\sum_{i=2}^{n}M(\lfloor n/i\rfloor),\qquad
\Phi(n)=\frac{n(n+1)}2-\sum_{i=2}^{n}\Phi(\lfloor n/i\rfloor),
\]
$M(n)=\sum_{i\le n}\mu(i)$，按等商段乘段长。先提升类型再乘；模下负和要归一化；
最后除以 2 时先整数除再取模，合数模下不擅自求 2 的逆元。

\section{整数分拆：重数限制的差一问题}\label{knowledge-partitions}\index{integer partitions}\index{pentagonal number theorem}
实现见第~\pageref{compact-Partitions}~页。$P(z)=\prod_{i\ge1}(1-z^i)^{-1}$，$p(0)=1$，五边形数定理
\[
\prod_{i\ge1}(1-z^i)=1+\sum_{j\ge1}(-1)^j
\left(z^{j(3j-1)/2}+z^{j(3j+1)/2}\right)
\]
给出 $p(n)$ 递推，符号 $+,+,-,-,\ldots$，每个 $n$ 有 $O(\sqrt n)$ 项；无需逆元，任意正模数（含 1）。
每个值至多出现 $k$ 次：
\[
\prod_{i\ge1}(1+z^i+\cdots+z^{ki})=P(z)\prod_{i\ge1}(1-z^{(k+1)i}),
\]
偏移放大 $k+1$ 倍，符号 $-,-,+,+,\ldots$。\texttt{limited(n,k)} 包含 $k$；$k=0$ 时非零 $n$ 无解，$k=1$ 为互异分拆。
“每种少于 $k$ 次”对应 \texttt{limited(n,k-1)}（$k\ge1$）。

\subsection{三个容易混淆的限制}
$A_k(n)$：部分均不超过 $k$ 的分拆数，等于至多 $k$ 个部分的分拆数，生成函数 $\prod_{i=1}^{k}(1-z^i)^{-1}$，
用面值 $1,\ldots,k$ 完全背包。恰好 $k$ 部分为 $A_k(n-k)$；恰好 $k$ 个互异部分为
$A_k\!\left(n-\frac{k(k+1)}2\right)$（用 \texttt{1LL*k*(k+1LL)/2}）。
负目标为 0；$A_0(0)=1$，$A_0(n)=0$（$n>0$）。这些都不同于限制每个值出现次数的 \texttt{limited}。

\subsection{重数限制与禁止倍数部分}
$d=k+1\ge1$ 时
\[
\prod_{i\ge1}\frac{1-z^{di}}{1-z^i}
=\prod_{\substack{i\ge1\\d\nmid i}}\frac1{1-z^i},
\]
即每值至多 $k$ 次等价于禁用 $d$ 的倍数，可用完全背包核对；$k=1$ 时互异分拆数等于奇数部分分拆数；$k\ge n$ 时无限制。
